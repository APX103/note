3322 空鼠手感终版

陀螺直出架构 · 延迟大追凶全案 · 全部定稿参数  ·  2026-10-14 · 系列第三篇（调研 → M0-M3 → M4完全指南 → 本篇·手感终版）· 代码：work/airmouse-3322 tag v0.1-feel-good

终态一句话：3322 + ICM20948 + EB21duino 三件套，上电即用，转手腕/摆臂屏幕光标实时跟随，拔插 dongle 自动重连，用户实测手感满意。信号链：陀螺−零偏 → 一阶滤波(τ≈1ms) → 静止门 → ×50px/deg → SLE(双端压缩连接参数) → USB HID——零姿态计算，CPU 5%，EKF 版留在备胎分支。

1 · 为什么放弃 EKF 姿态差分（三次滞后追凶的结果）

M3-M4 期间光标映射用的是"EKF 四元数→欧拉角→逐拍差分"。用户实测始终"非常不跟手"（0.5~1s 级滞后），追凶三轮：

#嫌疑判决依据
1EKF 加计观测滞后有罪（部分）快速转动时陀螺积分与低通后重力矢量"打架"，姿态输出永远慢半拍；叠加输入低通（0.1→0.5 后仍有 10ms+）与差分链
2motionless 冻结（导航级静止检测外泄到光标）有罪（大头之一）检测器的运动阈值 5°/s+迟滞窗口（为 ZUPT 设计）——小幅摆腕 2~3°/s 达不到解冻阈值，"摆了半秒才动"。修法：光标冻结交给死区（0.03°/拍≈3°/s 等效），motionless 只留给 EKF 内部
3SLE 传输慢无罪（但顺手修了）实测 100~1700ms"延迟"最终被证明是测量尺子的量化 bug（§3）；不过连接调度参数双端压缩后体感确实更好

结论（用户提出，实测验证）：空鼠不需要绝对姿态——陀螺角速度本身就是角位移的直接测量（∫ω·dt）。速率直出只欠一个零偏补偿，于是：

// 终版映射（sle_client.c，AIM_USE_GYRO_DIRECT=1）
corr = gyro − gyro_bias                 // 零偏：静止(<1°/s 持续0.3s)时 EMA(0.02) 缓慢跟踪
lpf  += 0.8×(corr − lpf)                // 一阶滤波 τ≈1ms
静止门内：输出 0 + 学零偏；否则：
dx = −ωz(dps) × 50 px/deg × 10ms        // 绕Z→dx（实测定稿，反号）
dy = +ωx(dps) × 50 px/deg × 10ms        // 绕X→dy（实测再反号定稿）

EKF（quat-zupt-ekf 七态）完整保留在固件里编译但不再驱动光标——stable/m4-gain50 分支可一键切回，未来做"姿态锚定/走廊混合"时再请它出山。

2 · 延迟大追凶全案（方法论教科书）

2.1 分段测量法

固件内埋"甩动打点"（角速度>150°/s 单发 [SNAP]），Mac 双端毫秒时间戳监听（3322 @750000 + dongle 日志口 @115200），拼时间线切三段：传感器+固件段 / SLE 空口段 / USB HID 段。甩动 5~12 次取分布。

2.2 三个反转

  1. 算法嫌疑：直出版一样"慢" → 算法洗清；
  2. 连接参数嫌疑：实测 100~1700ms 随机抖动，像"包等连接事件窗口"→ 双端压参数（见 §2.3），但复测依旧 13~1600ms；
  3. 尺子反转：发现 dongle 统计打印每 100 包一行——我监听计数跳变的"量化粒度"恰好 0~1s，与"延迟"分布完全重合。之前量的全是打印聚合周期，链路本身是快的。用户插回 USB HID 实测手感好 = 终审判决。
测量方法论铁律：聚合打印（每 N 包/行）做时间戳测量自带 0~N/rate 秒量化误差，必须用事件驱动的瞬时标记（ dongle 侧保留的 [BURST] 静→动首包瞬时打印就是修正后的尺子，零开销常驻）。另：两个瞬时标记必须语义对称（首包发送 ↔ 首包到达），错配（峰值时刻↔首包到达）会配对到下一次事件、秒级错位。

2.3 SLE 连接参数的正确姿势（踩坑实录）

做法结果
bs_sle_config(SLE_CONFIG_CONNECT_SET_DEFAULT_PARAM, ...) 全局默认PAIR 直接 timeout 破坏配对（回退验证实锤）
3322 侧：preset ADD（SLE_CONFIG_CONNECT_PRESET_PARAM，mode_id=1，interval 8-16 slot）+ 连接后 bs_sle_set_connect_param(dev_id, 1)（ssf_connection.h:860）✅ 不破坏配对
dongle 侧：连接回调 CONNECTED 分支 sle_update_connect_param（sle_connection_manager.h:979，server 主动发起，interval 8-16 slot）✅ 双管齐下

注意 SLE interval 单位是 slot（sle_connection_manager.h 注释），不是 BLE 的 1.25ms。

3 · 200Hz 实验（唯一一次算力翻车）

EKF 是 double 精度，RV32F 只有单精度硬件——200Hz 节拍下软浮点计算超 5ms/拍预算，CPU 饱和→看门狗复位循环（连心跳都没有），dongle 侧还叠半开连接致重连失败（0x80006016/0x8000600a）。回滚 100Hz 即愈。将来提速的正确路径：EKF 单精度 float 化（计算量降 5-10×，用 quat-zupt-ekf 差分框架重验证）——直出架构下已无此需求。

A2 · sle_client.c — IMU 驱动（init/read_regs/write_reg/bank/config）

static errcode_t aim_imu_init(void)
{
    uapi_pin_set_mode(4, 0); // SCL=MGPIO_4 mode0（AT+STARTI2CMASTER=...,4,5,0,0 实测 succ）
    uapi_pin_set_mode(5, 0); // SDA=MGPIO_5
    return uapi_i2c_master_init(AIM_IMU_BUS, 400000, 0);
}

// 组合事务：写寄存器地址后读 len 字节
static errcode_t aim_imu_read_regs(uint8_t reg, uint8_t *buf, uint16_t len)
{
    uint8_t addr = reg;
    uint8_t dummy[1] = {0};
    i2c_data_t d = {0};
    d.send_buf = &addr;
    d.send_len = 1;
    d.receive_buf = buf;
    d.receive_len = len;
    (void)dummy;
    return uapi_i2c_master_writeread(AIM_IMU_BUS, AIM_IMU_ADDR, &d);
}

// 纯写：寄寄存器
static errcode_t aim_imu_write_reg(uint8_t reg, uint8_t val)
{
    uint8_t w[2] = {reg, val};
    i2c_data_t d = {0};
    d.send_buf = w;
    d.send_len = 2;
    d.receive_buf = NULL;
    d.receive_len = 0;
    return uapi_i2c_master_write(AIM_IMU_BUS, AIM_IMU_ADDR, &d);
}

#define AIM_IMU_BANK_SEL 0x7F
static void aim_imu_switch_bank(uint8_t bank)
{
    (void)aim_imu_write_reg(AIM_IMU_BANK_SEL, bank);
}

// ICM20948 初始化：软复位→唤醒→六轴全开→量程/ODR（±4g / ±500dps / ~225Hz）
static errcode_t aim_imu_config(void)
{
    aim_imu_switch_bank(0);
    (void)aim_imu_write_reg(0x06, 0x80 | 0x41); // PWR_MGMT_1: DEVICE_RESET + SLEEP
    osal_msleep(60);                             // 复位后等待
    (void)aim_imu_write_reg(0x06, 0x01);        // PWR_MGMT_1: CLK_PLL，唤醒
    osal_msleep(5);
    (void)aim_imu_write_reg(0x07, 0x00);        // PWR_MGMT_2: 加计+陀螺全开
    aim_imu_switch_bank(2);
    (void)aim_imu_write_reg(0x01, 0x02);        // GYRO_CONFIG_1: ±500dps, DLPF on
    (void)aim_imu_write_reg(0x00, 0x04);        // GYRO_SMPLRT_DIV: 1125/(4+1)=225Hz
    (void)aim_imu_write_reg(0x14, 0x02);        // ACCEL_CONFIG: ±4g, DLPF on
    (void)aim_imu_write_reg(0x10, 0x00);        // ACCEL_SMPLRT_DIV_1
    (void)aim_imu_write_reg(0x11, 0x04);        // ACCEL_SMPLRT_DIV_2: 225Hz
    aim_imu_switch_bank(0);
    return ERRCODE_SUCC;
}

// 一拍：读 IMU→(EKF|直出)→输出 dx,dy（返回 false = 传感器读失败）
static bool aim_ekf_step(int16_t *dx_out, int16_t *dy_out)
{
    uint8_t raw[14] = {0};
    *dx_out = 0;
    *dy_out = 0;
    if (aim_imu_read_regs(AIM_IMU_DATA_REG, raw, 14) != ERRCODE_SUCC) {
        return false;
    }
    double acc[3], gyr[3];
    for (int i = 0; i < 3; i++) {
        int16_t a = (int16_t)((raw[i * 2] << 8) | raw[i * 2 + 1]);
        int16_t g = (int16_t)((raw[6 + i * 2] << 8) | raw[7 + i * 2 + 1]);
        acc[i] = (double)a / AIM_ACC_LSB_PER_G * AIM_G;          // → m/s²
        gyr[i] = (double)g / AIM_GYRO_LSB_DPS * AIM_D2R;         // → rad/s
    }
    double dx = 0;
    double dy = 0;
#if AIM_USE_GYRO_DIRECT
    // ---- 速率直出：零偏补偿 → 轻滤波 → 静止门 ----
    if (!g_aim_gyro_lpf_init) {
        for (int i = 0; i < 3; i++) {
            g_aim_gyro_lpf[i] = gyr[i];
        }
        g_aim_gyro_lpf_init = true;
    }
    double corr[3];
    for (int i = 0; i < 3; i++) {
        double c = gyr[i] - g_aim_gyro_bias_rad[i];
        g_aim_gyro_lpf[i] += AIM_GYRO_LPF_K * (c - g_aim_gyro_lpf[i]);
        corr[i] = g_aim_gyro_lpf[i];
    }
    double norm_dps = sqrt(corr[0] * corr[0] + corr[1] * corr[1] + corr[2] * corr[2]) / AIM_D2R;
    if (norm_dps < AIM_STILL_DPS) {
        g_aim_still_time += 1.0 / AIM_EKF_FS;
        if (g_aim_still_time >= AIM_STILL_HOLD_S) {
            // 静止：EMA 缓慢跟踪零偏，输出归零
            for (int i = 0; i < 3; i++) {
                g_aim_gyro_bias_rad[i] += AIM_BIAS_EMA * (gyr[i] - g_aim_gyro_bias_rad[i]);
            }
            g_aim_moved_px += 0;
            return true;
        }
    } else {
        g_aim_still_time = 0;
    }
    // 延迟分段测量打点：甩动(>150°/s)瞬间打印——对照甩动动作看 [SNAP] 出现延迟=传感器+固件段
    static uint32_t snap_armed = 1;
    if (snap_armed && norm_dps > 150.0) {
        printf("[SNAP] %d dps\n", (int)norm_dps);
        snap_armed = 0; // 单发：静止门重新充能后再武装
    }
    if (norm_dps < 10.0) {
        snap_armed = 1;
    }
    {
        double dps_z = corr[2] / AIM_D2R; // 绕 Z（deg/s）
        double dps_x = corr[0] / AIM_D2R; // 绕 X（deg/s）
        double dt = 1.0 / AIM_EKF_FS;
        dx = -dps_z * AIM_RATE_GAIN * dt; // 轴映射沿用实测定稿：绕Z→dx 反号
        dy = dps_x * AIM_RATE_GAIN * dt;  // 绕X→dy 正号
    }
#else
    ekf7_output o;
    ekf7_pipeline_calculate(&g_aim_ekf, acc, gyr, NULL, &o);     // 六轴模式
    // 轴映射（2026-10-14 用户实测校正）：绕 Z 轴转(偏航) → dx，绕 X 轴转(横滚) → dy，两轴均反号。
    double dyaw = (o.euler_rad[2] - g_aim_rpy_last[2]) / AIM_D2R;   // 绕 Z，deg
    double droll = (o.euler_rad[0] - g_aim_rpy_last[0]) / AIM_D2R;  // 绕 X，deg
    for (int i = 0; i < 3; i++) {
        g_aim_rpy_last[i] = o.euler_rad[i];
    }
    // 注意：不用 o.motionless 冻结输出（导航级判据的小幅迟滞问题，详见 motionless 注释）
    if (fabs(dyaw) < AIM_DEADBAND_DEG) {
        dyaw = 0;
    }
    if (fabs(droll) < AIM_DEADBAND_DEG) {
        droll = 0;
    }
    dx = -dyaw * AIM_RATE_GAIN;
    dy = droll * AIM_RATE_GAIN;
#endif
    if (dx > 127) { dx = 127; }
    if (dx < -127) { dx = -127; }
    if (dy > 127) { dy = 127; }
    if (dy < -127) { dy = -127; }
    *dx_out = (int16_t)dx;
    *dy_out = (int16_t)dy;
    g_aim_moved_px += (uint32_t)(fabs(dx) + fabs(dy));
    return true;
}

// 状态流：栈启用(一次) → SSAP注册(一次) → PAIR(全零地址直连，无需先扫) → 等连接 → FIND → RUN。
// 断线回 PAIR 态重试；FIND 结果由回调捕获 g_aim_write_handle。任务节拍 100Hz（10ms/拍）。
static void aim_conn_fsm(bool *need_delay)
{
    static uint8_t st = 0;
    static uint32_t wait = 0;
    static bool stack_inited = false;
    static bool g_aim_param_set_done = false;
    static bool ssap_inited = false;
    *need_delay = false;

    switch (st) {
        case 0: // 栈启用（一次）
            if (!stack_inited) {
                (void)at_sle_client_enable();
                aim_preset_conn_params(); // 仅 ADD preset；连接后由 case4 调 set（SET_DEFAULT 会破坏 PAIR）
                stack_inited = true;
            }
            wait = 300; // 等 3s 栈起
            st = 1;
            break;
        case 1:
            if (--wait == 0) {
                if (!ssap_inited) {
                    (void)sle_demo_client_init(); // 注册 client + 拉起本任务(幂等)
                    ssap_inited = true;
                }
                st = 2;
            }
            break;
        case 2: { // 发起配对（全零地址直连广播中的 dongle）
            if (bt_sle_is_connected()) {
                st = 4;
                break;
            }
            sle_pair addr = {0};
            (void)at_client_connect_to_peripheral(&addr);
            wait = 1500; // 15s 超时
            st = 3;
            break;
        }
        case 3: // 等连接
            if (bt_sle_is_connected()) {
                printf("[AIM][FSM] connected\n");
                st = 4;
            } else if (--wait == 0) {
                st = 2; // 超时重试
            }
            break;
        case 4: // 已连接：压连接间隔 → 缺句柄则 FIND
            if (!bt_sle_is_connected()) {
                g_aim_write_handle = 0; // 断线：清句柄回配对态
                g_aim_param_set_done = false;
                st = 2;
                break;
            }
            if (!g_aim_param_set_done) { // 连接后一次性压间隔（治传输抖动）
                errcode_t pr = bs_sle_set_connect_param(g_client_device_id, 1);
                printf("[AIM][FSM] set_connect_param: 0x%x\n", pr);
                g_aim_param_set_done = true;
            }
            if (g_aim_write_handle == 0) {
                (void)sle_ssapc_find_all_service(g_client_device_id, g_client_id, false);
                wait = 300;
                st = 5;
            }
            break;
        case 5: // 等 FIND 回调捕获句柄
            if (g_aim_write_handle != 0) {
                printf("[AIM][FSM] run (handle=0x%04x)\n", g_aim_write_handle);
                st = 4;
            } else if (--wait == 0) {
                st = 4; // 再触发一次 FIND
            }
            break;
        default:
            st = 0;
            break;
    }
}

static int aim_demo_task(void *arg)
{
    uint32_t sent = 0;
    uint32_t fail = 0;
    uint32_t skipped = 0;
    uint32_t imu_fail = 0;
    uint32_t stat_div = 0;
    (void)arg;
    errcode_t imu_ret = aim_imu_init();
    printf("[AIM] imu init: 0x%x\n", imu_ret);
    aim_imu_config(); // 软复位+唤醒+量程/ODR
    aim_imu_report(); // 开机即读一次
    ekf7_pipeline_init(&g_aim_ekf, AIM_EKF_FS, 1, 0); // work_mode=1（验证态一致），六轴
    g_aim_ekf_init = true;
    // 量程寄存器回读（判断 bank2 配置是否生效，决定刻度）
    uint8_t gcfg = 0, acfg = 0;
    aim_imu_switch_bank(2);
    (void)aim_imu_read_regs(0x01, &gcfg, 1);  // GYRO_CONFIG_1
    (void)aim_imu_read_regs(0x14, &acfg, 1);  // ACCEL_CONFIG
    aim_imu_switch_bank(0);
    printf("[AIM][IMU] gyro_cfg=0x%02X accel_cfg=0x%02X (FS bits: 0=±250/2g 2=±500/4g)\n", gcfg, acfg);
    printf("[AIM] demo task ready (EKF dx/dy @%gHz, gain=%g px/deg, auto-connect)\n", AIM_EKF_FS, AIM_RATE_GAIN);
    while (1) {
        osal_msleep(AIM_DEMO_TICK_MS);
        // 自动连接状态机（每拍推进）
        bool dummy = false;
        aim_conn_fsm(&dummy);
        // EKF 无条件每拍跑：姿态解算独立于 SLE 链路（调算法不被连接状态绑架）
        int16_t dx_now = 0;
        int16_t dy_now = 0;
        if (!aim_ekf_step(&dx_now, &dy_now)) {
            imu_fail++;
        }
        if (++stat_div >= AIM_EKF_FS) { // 每秒打印姿态与统计
            stat_div = 0;
            double rpy[3];
            ekf7_filter_euler(&g_aim_ekf.filt, rpy);
            printf("[AIM] rpy=[%.1f %.1f %.1f]deg moved=%upx sent=%u imu_fail=%u align=%.1f/%.0f conn=%d hdl=0x%x skip=%u\n",
                rpy[0] / AIM_D2R, rpy[1] / AIM_D2R, rpy[2] / AIM_D2R,
                g_aim_moved_px, sent, imu_fail,
                g_aim_ekf.filt.init_max_time, g_aim_ekf.filt.flag_in,
                (int)bt_sle_is_connected(), (unsigned)g_aim_write_handle, skipped);
        }
        if (!bt_sle_is_connected() || g_aim_write_handle == 0) {
            continue; // 未连接/未发现服务时静默等待
        }
        // 退避背压（无确认写模式）：连续失败=栈队列压力，停 20 拍（200ms）再试
        if (g_aim_backoff > 0) {
            g_aim_backoff--;
            skipped++;
            continue;
        }
        // 发送：直接用本拍 EKF 输出（零位移不发）
        aim_mouse_pkt_t pkt;
        pkt.magic = 0xA5;
        pkt.keys = 0;
        pkt.dx = dx_now;
        pkt.dy = dy_now;
        if (pkt.dx == 0 && pkt.dy == 0) {
            continue;
        }
        if (sle_ssapc_write_buf(g_client_device_id, g_client_id, g_aim_write_handle,
            (const uint8_t *)&pkt, sizeof(pkt)) == ERRCODE_BT_SUCCESS) {
            sent++;
            g_aim_fail_streak = 0;
        } else {
            fail++;
            if (++g_aim_fail_streak >= 8) { // 连续8败=队列满，退避
                g_aim_backoff = 20;
                g_aim_fail_streak = 0;
            }
        }
        if ((sent + fail + skipped) % 500 == 0) {
            printf("[AIM] stream: sent=%u fail=%u skip=%u bk=%u\n", sent, fail, skipped, g_aim_backoff);
        }
    }
    return 0;
}
